                25@ > 0
            then
                for 26@ = 0 to 6

                        003B: 29@ == 27@
                    then
                        return
                    end
                end
            end

            // Exclude Special Actors ONLY during missions (Slots 1 to 10: Models 290..299)
            if
                0E1D: is_on_mission
            then
                if and
                    5@ >= 290
                    299 >= 5@
                then
                    return
                end
            end

            if
                8449: not actor 29@ in_a_car // Strictly On Foot (No passenger glitches!)
            then
                // Determine if target should retaliate with bare fists upon standing up
                24@ = 0

                // Check if ped was actively targeting player before blast
                0E44: get_char_kill_target_char 29@ store_to 23@

                    003B: 23@ == $PLAYER_ACTOR
                then
                    24@ = 1
                end

                // Check if target is an enemy gang member during a live Gang War
                if
                    087A: gang_war_in_progress
                then
                    089F: get_actor 29@ pedtype_to 23@
                    if and
                        23@ >= 7
                        14 >= 23@
                    then
                        if
                            8039: not 23@ == 8 // Exclude Grove Street
                        then
                            24@ = 1
                        end
                    end
                end

                // Sever ambient gang group hierarchy so group AI cannot override tasks
                06C9: remove_actor 29@ from_group

                // Kill all weapons, tasks, and forward momentum
                048F: actor 29@ remove_weapons
                0687: clear_actor 29@ task
                0E96: clear_char_primary_tasks 29@
                0E97: clear_char_secondary_tasks 29@
                083C: set_actor 29@ velocity_in_direction_XYZ 0.0 0.0 0.0

                0851: set_actor 29@ decrease_health_by 5 affect_armour 0

                // Force immediate bone-level animation snap to override gang walk style

                // Queue Task Sequence: Skid -> Get Up -> Bare-fisted Retaliation if Hostile

                0812: AS_actor -1 perform_animation "KO_SKID_BACK" IFP "PED" framedelta 8.0 loopA 0 lockX 1 lockY 1 lockF 1 time 1500

                    24@ == 1
                then
                    05E2: AS_actor -1 kill_actor $PLAYER_ACTOR
                end
                0616: define_AS_pack_end 2@

                061B: remove_references_to_AS_pack 2@
