{$CLEO .c}
{$USE CLEO+}
{$USE ini}
{$USE memory}

0000: NOP
script_name "Y_DebtD"

wait 1000

// =========================================================================
// YORIGAMI HEAP DICTIONARY (SKINNED)
// =========================================================================
const
    MEM_SAVE_SLOT = 0             // Formerly 700
    MEM_ALLOC_DEBTDET = 100       // Formerly 755
end

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    0002: jump @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
// 31@ NOW HOLDS THE MASTER POINTER.

1@ = -999 
12@ = 0 // Detonation Cooldown Remaining (ms)

:YDebtDet_StartupWait
wait 250
if
    8256: not is_player_playing $PLAYER_CHAR
then
    0002: jump @YDebtDet_StartupWait
end

:YDebtDet_Loop
wait 100

if
    8256: not is_player_playing $PLAYER_CHAR
then
    0002: jump @YDebtDet_Loop
end

01F5: $PLAYER_ACTOR = get_player_actor $PLAYER_CHAR

// Check if Save Slot Changed
0085: 30@ = 31@
000A: 30@ += MEM_SAVE_SLOT
0A8D: 0@ = read_memory 30@ size 4 virtual_protect 0
if
    803B: not 0@ == 1@
then
    gosub @YDebtDet_ReloadSettings
end

// Read Live Allocated Points in Debt Detonation (Shion Perk #6)
0085: 30@ = 31@
000A: 30@ += MEM_ALLOC_DEBTDET
0A8D: 17@ = read_memory 30@ size 4 virtual_protect 0

// Tick down Detonation Cooldown
if
    12@ > 0
then
    000E: 12@ -= 100
    if
        8019: not 12@ > 0
    then
        12@ = 0
    end
end

// -------------------------------------------------------------------------
// DEBT DETONATION TRIGGER
// -------------------------------------------------------------------------
if and
    17@ > 0     // Has points in Debt Detonation
    8019: not 12@ > 0    // Not on cooldown
    816B: not fading
    8E25: not is_on_cutscene
    8EB7: not is_on_scripted_cutscene
then
    0226: 26@ = actor $PLAYER_ACTOR health
    0653: 18@ = get_float_stat 24
    0017: 18@ /= 5.69
    0092: 16@ = float 18@ to_integer

    if
        16@ <= 0
    then
        16@ = 100
    end

    0085: 27@ = 26@
    0012: 27@ *= 100
    0072: 27@ /= 16@

    if and
        26@ > 0
        002D: 10@ >= 27@ // Health % <= Threshold
    then
        // Launch Shared Spirit Flash Shockwave!
        0A92: create_custom_thread "Yorigami\FX_GrimBurst.c"

        // Dynamic Cooldown Calculation
        0AF0: 12@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "BaseCooldown"
        if
            12@ == 0
        then
            12@ = 60
        end

        0AF0: 23@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "CooldownReductionPerPoint"
        if
            23@ == 0
        then
            23@ = 2
        end

        0085: 24@ = 17@
        006A: 24@ *= 23@ // Local-to-Local Integer Multiply (006A)
        0062: 12@ -= 24@

        0AF0: 23@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "MinDetonationCooldown"
        if
            23@ == 0
        then
            23@ = 10
        end

        if
            001D: 23@ > 12@
        then
            0085: 12@ = 23@
        end
        0012: 12@ *= 1000 // Convert seconds to ms

        0003: shake_camera 120
        if
            14@ == 1
        then
            018C: play_sound 1052 at 0.0 0.0 0.0
        end
        if
            11@ == 1
        then
            0ADF: add_text_label "Y_DDET" text "~r~MISFORTUNE DETONATION~n~~w~CATASTROPHE UNLEASHED"
            00BA: show_text_styled GXT 'Y_DDET' time 2500 style 5
        end

        // Dynamic Radius Calculation
        0AF2: 13@ = read_float_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "BaseRadius"
        if
            13@ <= 0.0
        then
            13@ = 30.0
        end

        0AF2: 25@ = read_float_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "RadiusPerPoint"
        if
            25@ <= 0.0
        then
            25@ = 1.5
        end

        0093: 24@ = integer 17@ to_float
        006B: 24@ *= 25@
        005B: 13@ += 24@

        0AF2: 25@ = read_float_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "MaxRadiusCap"
        if
            25@ <= 0.0
        then
            25@ = 60.0
        end

        if
            0025: 13@ > 25@
        then
            0087: 13@ = 25@
        end

        0A92: create_custom_thread "Yorigami\DisarmShockwave.c" 13@
    end
end

0002: jump @YDebtDet_Loop


// =========================================================================
// SUBROUTINES
// =========================================================================

:YDebtDet_ReloadSettings
0085: 1@ = 0@

0AF0: 10@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "HealthPercentThreshold"
if
    10@ == 0
then
    10@ = 15
end

0AF0: 11@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "ShowDetonationMessage"
0AF0: 14@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "DebtDetonation" key "PlayDetonationSound"
return
