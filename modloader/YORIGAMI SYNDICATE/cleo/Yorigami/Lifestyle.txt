{$CLEO .c}
{$USE CLEO+}
{$USE ini}
{$USE memory}

0000: NOP
script_name "Y_Life"

wait 1000

// =========================================================================
// YORIGAMI MASTER HEAP DICTIONARY (SKINNED FOR LIFESTYLE)
// =========================================================================
const
    MEM_SAVE_SLOT = 0
    MEM_ALLOC_TAXIEXT = 44
    MEM_ALLOC_DRIPWORSHIP = 48
    MEM_ALLOC_MIDAS = 56
end

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
if
    31@ == 0
then
    goto @Connect_To_Brain
end
// 31@ NOW HOLDS THE MASTER POINTER.

1@ = -999

:YLifestyle_StartupWait
wait 250
if
    8256: not player $PLAYER_CHAR defined
then
    goto @YLifestyle_StartupWait
end

:YLifestyle_Loop
wait 100

if
    8256: not player $PLAYER_CHAR defined
then
    goto @YLifestyle_Loop
end
01F5: $PLAYER_ACTOR = get_player_actor $PLAYER_CHAR

// Check if Save Slot Changed
0085: 30@ = 31@
000A: 30@ += MEM_SAVE_SLOT
0A8D: 30@ = read_memory 30@ size 4 virtual_protect 0
if
    803B: not 30@ == 1@
then
    0085: 1@ = 30@
    gosub @YLifestyle_ReloadSettings
end

// Read Live Allocated Perks from Master Memory
0085: 30@ = 31@
000A: 30@ += MEM_ALLOC_TAXIEXT
0A8D: 2@ = read_memory 30@ size 4 virtual_protect 0

0085: 30@ = 31@
000A: 30@ += MEM_ALLOC_DRIPWORSHIP
0A8D: 3@ = read_memory 30@ size 4 virtual_protect 0

0085: 30@ = 31@
000A: 30@ += MEM_ALLOC_MIDAS
0A8D: 4@ = read_memory 30@ size 4 virtual_protect 0

// -------------------------------------------------------------------------
// DRIP WORSHIP (Memory Stat Injection)
// -------------------------------------------------------------------------
if
    3@ > 0
then
    0085: 26@ = 3@
    006A: 26@ *= 10@ // Points * PercentPerPoint (006A)
    0012: 26@ *= 10  // Scale 0..1000
    if
        26@ > 1000
    then
        26@ = 1000
    end
    0093: 9@ = integer 26@ to_float
    0A8C: write_memory 12030948 size 4 value 9@ virtual_protect 0
    if
        26@ >= 1000
    then
        0A8C: write_memory 9867648 size 1 value 1 virtual_protect 0
    else
        0A8C: write_memory 9867648 size 1 value 0 virtual_protect 0
    end
else
    0A8C: write_memory 9867648 size 1 value 0 virtual_protect 0
end

// -------------------------------------------------------------------------
// TAXI EXTORTION BONUS (Active if Alloc_TaxiExtortion > 0)
// -------------------------------------------------------------------------
if
    2@ > 0
then
    0652: 26@ = integer_stat 150
    0085: 27@ = 26@
    0062: 27@ -= 6@
    if
        27@ > 0
    then
        0652: 27@ = integer_stat 149
        0085: 28@ = 27@
        0062: 28@ -= 7@

        0085: 27@ = 2@
        006A: 27@ *= 8@ // Points * BonusPercentPerPoint (006A)

        0085: 26@ = 28@
        006A: 26@ *= 27@ // Base * Bonus % (006A)
        0016: 26@ /= 100

        if
            26@ > 0
        then
            0109: player $PLAYER_CHAR money += 26@
            if
                14@ == 1
            then
                018C: play_sound 1058 at 0.0 0.0 0.0
            end
            if
                5@ == 1
            then
                0ADF: add_text_label "Y_TAXI" text "~y~EXTORTION BONUS~n~~g~+$~1~"
                01E3: show_text_1number_styled GXT 'Y_TAXI' number 26@ time 2500 style 5
            end
        end

        0652: 6@ = integer_stat 150
        0652: 7@ = integer_stat 149
    end
else
    0652: 6@ = integer_stat 150
    0652: 7@ = integer_stat 149
end

// -------------------------------------------------------------------------
// MIDAS MAGNET (Active if Alloc_MidasMagnet > 0)
// -------------------------------------------------------------------------
if
    4@ > 0
then
    // Calculate Dynamic Suction Radius: BaseRadius + (Points * RadiusPerPoint)
    0093: 23@ = integer 4@ to_float
    006B: 23@ *= 16@ // Points * RadiusPerPoint (float)
    005B: 23@ += 15@ // Add BaseRadius (float)
    if
        0025: 23@ > 17@ // Clamp to MaxRadiusCap (float var-to-var)
    then
        0087: 23@ = 17@
    end

    // Calculate Dynamic Cash Bonus %: Points * BonusPercentPerPoint (int)
    0085: 24@ = 4@
    006A: 24@ *= 18@ // Points * BonusPercentPerPoint (int)
    if
        001D: 24@ > 19@ // Clamp to MaxBonusPercentCap (int var-to-var)
    then
        0085: 24@ = 19@
    end

    // Player Coordinates
    00A0: store_actor $PLAYER_ACTOR position_to 20@ 21@ 22@

    // Batch Accumulators
    25@ = 0 // Total Cash Collected this tick
    26@ = 0 // Count of Pickups vacuumed this tick

    // Scan GTA SA Pickup Pool (0x9788C0 to 0x97D620, step 32 bytes)
    for 11@ = 0x9788C0 to 0x97D620 step 0x20
        0085: 30@ = 11@
        000A: 30@ += 0x1C
        0A8D: 30@ = read_memory 30@ size 1 virtual_protect 0

        // Type 8 = Cash Pickup, Type 19 = Cash Pickup Doesn't Disappear
        if or
            30@ == 8
            30@ == 19
        then
            // Validate physical existence in the world (m_pObject != 0 at +0x04)
            0085: 30@ = 11@
            000A: 30@ += 0x04
            0A8D: 30@ = read_memory 30@ size 4 virtual_protect 0
            if
                8039: not 30@ == 0
            then
                // Calculate Script Handle
                0085: 30@ = 11@
                000A: 30@ += 0x1A
                0A8D: 30@ = read_memory 30@ size 2 virtual_protect 0
                0012: 30@ *= 65536
                0085: 29@ = 11@
                000E: 29@ -= 0x9788C0
                0016: 29@ /= 0x20
                005A: 30@ += 29@ // 30@ = Pickup Script Handle

                // Get Pickup World Coordinates
                065B: store_pickup 30@ position_to 27@ 28@ 29@

                // Check distance to CJ
                050A: 9@ = distance_between_XYZ 20@ 21@ 22@ and_XYZ 27@ 28@ 29@
                if
                    8025: not 9@ > 23@ // Within dynamic suction radius! (float var-to-var)
                then
                    // Read Cash Value (offset +0x08) BEFORE destroying pickup
                    0085: 9@ = 11@
                    000A: 9@ += 0x08
                    0A8D: 9@ = read_memory 9@ size 4 virtual_protect 0 // 9@ = Base Cash Value

                    if
                        9@ <= 0
                    then
                        9@ = 50
                    end

                    // Destroy pickup from world immediately
                    0215: destroy_pickup 30@

                    // Calculate Pure Bonus Cash into 30@ (Strict 006A)
                    0085: 30@ = 9@
                    006A: 30@ *= 24@ // Base * Bonus % (006A)
                    0016: 30@ /= 100 // 30@ = Pure Bonus Cash

                    // Add to Batch Accumulators
                    005A: 25@ += 9@  // Total Cash += Base
                    005A: 25@ += 30@ // Total Cash += Bonus
                    000A: 26@ += 1   // Pickup Count += 1

                    // Optional Particle Burst at Pickup Location (Auto-Cleans!)
                    if
                        0@ == 1
                    then
                        000B: 29@ += 0.35 // Lift slightly off ground
                        064B: 30@ = create_particle "joon_gold" at 27@ 28@ 29@ type 1
                        064C: make_particle 30@ visible
                        064F: remove_references_to_particle 30@
                    end
                end
            end
        end
    end

    // Award Batch Total
    if
        26@ > 0
    then
        0109: player $PLAYER_CHAR money += 25@

        if
            13@ == 1
        then
            018C: play_sound 1058 at 0.0 0.0 0.0
        end

        if
            12@ == 1
        then
            if
                26@ == 1
            then
                0ADF: add_text_label "Y_MIDAS" text "~y~MIDAS MAGNET~n~~g~+$~1~"
                01E3: show_text_1number_styled GXT 'Y_MIDAS' number 25@ time 2000 style 5
            else
                0ADF: add_text_label "Y_MDAS2" text "~y~MIDAS MAGNET~n~~g~+$~1~ ~w~(~1~ STACKS)"
                036D: show_text_2numbers_styled GXT 'Y_MDAS2' numbers 25@ 26@ time 2000 style 5
            end
        end
    end
end

goto @YLifestyle_Loop

    15@ <= 0.0
then
    15@ = 8.0
end

0AF2: 16@ = read_float_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "RadiusPerPoint"
if
    16@ <= 0.0
then
    16@ = 0.5
end

0AF2: 17@ = read_float_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "MaxRadiusCap"
if
    17@ <= 0.0
then
    17@ = 35.0
end

0AF0: 18@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "BonusPercentPerPoint"
if
    18@ < 0
then
    18@ = 0
end

0AF0: 19@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "MaxBonusPercentCap"
if
    19@ <= 0
then
    19@ = 100
end

0AF0: 12@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "ShowVacuumMessage"
0AF0: 13@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "PlayChimeSound"
0AF0: 0@ = read_int_from_ini_file "cleo\YorigamiPerks.ini" section "MidasMagnet" key "SpawnSparks"

0652: 6@ = integer_stat 150
0652: 7@ = integer_stat 149
