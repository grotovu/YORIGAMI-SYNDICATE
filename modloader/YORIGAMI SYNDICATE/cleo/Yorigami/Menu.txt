// =========================================================================

        0AB0: is_key_pressed 17@ // M Key

                18@ = 0

                001D: 6@ > 1@
            then
                0085: 6@ = 1@

            005A: 4@ += 24@

            000A: 30@ += MEM_AVAIL_POINTS

                0019: 6@ > 100
            then

                018C: play_sound 1058 at 0.0 0.0 0.0

            018C: play_sound 1057 at 0.0 0.0 0.0
