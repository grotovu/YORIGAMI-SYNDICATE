                                        08B7: test 3@ bit 2

                                    0085: 27@ = 26@
                                    0016: 27@ /= 2

                                        056D: actor 29@ defined
                                        8118: not actor 29@ dead

                                        then
                                            01C2: remove_references_to_actor 29@

                                    000A: 30@ += MEM_LAST_EXTORTED_PED

                                        0AE1: 0@ = random_char_near_point 21@ 22@ 23@ in_radius 18.0 find_next 24@ pass_deads 1

                                                0039: 25@ == 6

                                                    0039: 25@ == 0

        end
    end
end
