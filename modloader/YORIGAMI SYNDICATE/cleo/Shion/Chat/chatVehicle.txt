// =========================================================================
// YORIGAMI MASTER HEAP DICTIONARY (PRUNED FOR SHIONVEHICLE)
// =========================================================================
const
    MEM_PRIORITY_LOCK = 20        // Priority lock latch
    MEM_SHION_HANDLE = 116        // Shion ped handle (-1 = Hitodama, >0 = Physical)
    MEM_SHION_PRES_HP = 160       // Hitodama presence HP / flame intensity
end

:Connect_To_Brain
wait 0 ms
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
if
    31@ == 0
then
    goto @Connect_To_Brain
end
// 31@ NOW HOLDS THE MASTER POINTER SAFELY.

// Check if Tier 3 (Action) or Tier 4 (Theater) is active
0085: 30@ = 31@
000A: 30@ += MEM_PRIORITY_LOCK
0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
if
    04B5: 19@ >= 3
then
    0A93: terminate_this_custom_script
end

// Preempt lower-tier idle chatter scripts
0ABA: terminate_all_custom_scripts_with_this_name 'Sh_Zone'

// Claim Tier 2 Vehicle Lock
0085: 30@ = 31@
000A: 30@ += MEM_PRIORITY_LOCK
0A8C: write_memory 30@ size 4 value 2 virtual_protect 0

// Validate Player Life & Driving State
if
    8256: not player $PLAYER_CHAR defined
then
    goto @Finalize_Ambient_Chat
end

01F5: $PLAYER_ACTOR = get_player_actor $PLAYER_CHAR
if or
    0118: actor $PLAYER_ACTOR dead
    0741: actor $PLAYER_ACTOR busted
    80DF: not actor $PLAYER_ACTOR driving
then
    goto @Finalize_Ambient_Chat
end

// 0@ = Shion Actor Handle passed from ShionChat.c
0085: 10@ = 0@

03C0: 24@ = actor $PLAYER_ACTOR car
0441: 25@ = car 24@ model

// Check Form Mode: 0 = Physical Ped, 1 = Hitodama Spirit
0085: 30@ = 31@
000A: 30@ += MEM_SHION_HANDLE
0A8D: 10@ = read_memory 30@ size 4 virtual_protect 0
if
    10@ == -1
then
    11@ = 1 // Hitodama Form
else
    11@ = 0 // Physical Ped Form
end

gosub @Check_Alive_Or_Exit

// =========================================================================
// PHYSICAL FORM VEHICLE PRESENCE & DISTANCE GUARDS
// =========================================================================
if
    11@ == 0
then
    if
        10@ > 0
    then
        if
            056D: actor 10@ defined
        then
            // Bicycles have 1 seat: Shion must be sprinting nearby on foot
            if or
                25@ == 481
                25@ == 509
                25@ == 510
            then
                if
                    8EE4: not locate_char_distance_to_char $PLAYER_ACTOR char 10@ radius 30.0
                then
                    goto @Finalize_Ambient_Chat
                end
            else
                // All other vehicles (Motorcycles, Cars, Aircraft, Boats): Must be passenger!
                if
                    80DB: not actor 10@ in_car 24@
                then
                    goto @Finalize_Ambient_Chat
                end
            end
        else
            goto @Finalize_Ambient_Chat
        end
    end
end
