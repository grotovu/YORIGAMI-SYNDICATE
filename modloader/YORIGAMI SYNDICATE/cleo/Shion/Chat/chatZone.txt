const
    MEM_PRIORITY_LOCK = 20        // Lock channel to prevent chatter overlap
    MEM_SHION_HANDLE = 116        // Shion actor handle or -1 if Hitodama

:Connect_To_Brain
0001: wait 0 ms
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0
then
    jump @Connect_To_Brain
end
000A: 31@ += 0x3C
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0
// 31@ NOW HOLDS THE MASTER POINTER.

// Check if any higher priority tier is already holding the channel
0085: 30@ = 31@
000A: 30@ += MEM_PRIORITY_LOCK
0A8D: 16@ = read_memory 30@ size 4 virtual_protect 0
if
    16@ >= 3
then
    0A93: terminate_this_custom_script
end

// Claim Tier 2 Ambient Lock
0085: 30@ = 31@
000A: 30@ += MEM_PRIORITY_LOCK
0A8C: write_memory 30@ size 4 value 2 virtual_protect 0

gosub @Check_Alive_Or_Exit

// Check Form Mode: 0 = Physical Ped, 1 = Hitodama Spirit
0085: 30@ = 31@
000A: 30@ += MEM_SHION_HANDLE
0A8D: 10@ = read_memory 30@ size 4 virtual_protect 0
if
    10@ == -1
then
    11@ = 1 // Hitodama Form
else
    11@ = 0 // Physical Ped Form
end

// Read Anti-Repetition Location Memory from Shared Heap
0085: 30@ = 31@
000A: 30@ += MEM_LAST_ZONE_INT
