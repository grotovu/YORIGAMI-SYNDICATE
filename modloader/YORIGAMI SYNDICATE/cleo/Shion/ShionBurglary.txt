{$CLEO .c}
{$USE CLEO+}
{$USE memory}

0000: NOP
script_name "Sh_Burg"

// =========================================================================
// YORIGAMI MASTER HEAP DICTIONARY (SCOPED TO SHION BURGLARY)
// =========================================================================
const
    MEM_SHION_BUSY = 4            // Formerly 708
    MEM_PRIORITY_LOCK = 20        // Formerly 765
end

// 0@ = Shion Actor Handle passed on boot
0085: 10@ = 0@

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    0039: 31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
if
    0039: 31@ == 0
then
    goto @Connect_To_Brain
end
// 31@ NOW HOLDS THE MASTER POINTER.

1@ = 0  // Runner State: 0 = Following Jo'on, 1 = Away at Van
2@ = 0  // Van Run Timer (ms)
3@ = -1 // Anti-Repetition: Last Handoff Line Index
4@ = -1 // Anti-Repetition: Last Return Line Index
11@ = 0 // Carried Object Handle
12@ = 0 // TAB Key Hold Timer (ms)
13@ = 0 // Return Subtitle Lock Release Timer (ms)

:ShionBurglary_MainLoop
wait 50

// Non-blocking Return Subtitle Lock Release Timer
if
    0019: 13@ > 0
then
    000E: 13@ -= 50
    if
        8019: not 13@ > 0
    then
        13@ = 0
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
        if
            0039: 19@ == 3
        then
            0085: 30@ = 31@
            000A: 30@ += MEM_PRIORITY_LOCK
            0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
        end
    end
end

if or
    8256: not is_player_playing $PLAYER_CHAR
    8019: not 10@ > 0
    856D: not actor 10@ defined
    0118: actor 10@ dead
then
    gosub @Restore_Shion_Burglary_State
    0A93: terminate_this_custom_script
end

// Only active during Burglary Mission ($ONMISSION_BURGLARY == 1)
if
    8038: not $ONMISSION_BURGLARY == 1
then
    if or
        0019: 1@ > 0
        0019: 13@ > 0
    then
        gosub @Restore_Shion_Burglary_State
    end
    wait 500
    goto @ShionBurglary_MainLoop
end

// Alarm / Wanted Bailout
if or
    0038: $BURGLARY_DETECTED_FLAG == 1
    010F: player $PLAYER_CHAR wanted_level > 0
then
    if or
        0019: 1@ > 0
        0019: 13@ > 0
    then
        gosub @Restore_Shion_Burglary_State
    end
    wait 250
    goto @ShionBurglary_MainLoop
end

077E: get_active_interior_to 28@

// =========================================================================
// STATE 0: SHION FOLLOWS JO'ON & MONITORS FOR [TAB] HAND-OFF
// =========================================================================
if
    0039: 1@ == 0
then
    // Must be inside a house on foot
    if and
        0019: 28@ > 0
        80DF: not actor $PLAYER_ACTOR driving
        8449: not actor 10@ in_a_car
        816B: not fading
    then
        // Read active carried object handle (49@ in BURGJB memory at 0xA48A24)
        0A8D: 11@ = read_memory 0xA48A24 size 4 virtual_protect 0

        // Check if Jo'on is actively carrying an item
        if and
            0019: 11@ > 0
            03CA: object 11@ exists
            0737: actor $PLAYER_ACTOR lifting_object 11@
        then
            // Check distance between Jo'on and Shion
            00A0: store_actor $PLAYER_ACTOR position_to 20@ 21@ 22@
            00A0: store_actor 10@ position_to 23@ 24@ 25@
            050A: 26@ = distance_between_XYZ 20@ 21@ 22@ and_XYZ 23@ 24@ 25@

            // Close proximity (within 2.2m)
            if
                8021: not 26@ > 2.2
            then
                // Style 5 prompt: Right-side banner, leaves subtitle channel open
                0ADF: add_text_label "Y_BGHD" text "~y~BURGLARY MULE~n~~w~HOLD [TAB] TO DUMP LOOT ON SHION"
                00BA: show_text_styled GXT 'Y_BGHD' time 100 style 5

                // Check TAB / Action Key
                if or
                    00E1: player 0 pressed_key 4
                    0AB0: is_key_pressed 9 // VK_TAB
                then
                    000A: 12@ += 50
                    if
                        0029: 12@ >= 350 // Held for 0.35s
                    then
                        12@ = 0
                        gosub @Execute_Loot_Handoff
                    end
                else
                    12@ = 0
                end
            else
                12@ = 0
            end
        else
            12@ = 0
        end
    else
        12@ = 0
    end
end

// =========================================================================
// STATE 1: SHION AT THE VAN (TIMED SPIRIT RUN & DEPOSIT)
// =========================================================================
if
    0039: 1@ == 1
then
    000A: 2@ += 50

    // After 3 seconds, deposit into Boxville and credit mission memory!
    if
        0029: 2@ >= 3000
    then
        11@ = 0

        // INJECT +1 INTO BURGJB MISSION MEMORY (48@ at 0xA48A20)
        0A8D: 20@ = read_memory 0xA48A20 size 4 virtual_protect 0
        000A: 20@ += 1
        0A8C: write_memory 0xA48A20 size 4 value 20@ virtual_protect 0

        // Play native cash register chime
        018C: play_sound 1058 at 0.0 0.0 0.0

        // Release the handoff lock from 3 seconds ago
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
        if
            0039: 19@ == 3
        then
            0085: 30@ = 31@
            000A: 30@ += MEM_PRIORITY_LOCK
            0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
        end

        // Confirmation line: 10% chance to speak on return (with anti-repetition)
        if
            0E4D: random_percent 10
        then
            0209: 26@ = random_int_in_ranges 0 3
            if
                003B: 26@ == 4@
            then
                0209: 27@ = random_int_in_ranges 1 3
                005A: 26@ += 27@
                if
                    0029: 26@ >= 3
                then
                    000E: 26@ -= 3
                end
            end
            0085: 4@ = 26@

            03EB: clear_small_messages_only

            if
                0039: 26@ == 0
            then
                0ADF: add_text_label "Y_BGRT1" text "~b~Shion: ~s~\"Dumped it in the back... my arms feel like jelly, Jo'on. We'd better be almost done.\""
                09C1: add_next_text_to_brief_history 1
                00BC: show_text_highpriority GXT 'Y_BGRT1' time 3000 flag 1
            end
            if
                0039: 26@ == 1
            then
                0ADF: add_text_label "Y_BGRT2" text "~b~Shion: ~s~\"It's in the van. If we trip an alarm, I'm hiding behind you.\""
                09C1: add_next_text_to_brief_history 1
                00BC: show_text_highpriority GXT 'Y_BGRT2' time 3000 flag 1
            end
            if
                0039: 26@ == 2
            then
                0ADF: add_text_label "Y_BGRT3" text "~b~Shion: ~s~\"Stashed it. My back's screaming, Jo'on... you promised a full meal after this run.\""
                09C1: add_next_text_to_brief_history 1
                00BC: show_text_highpriority GXT 'Y_BGRT3' time 3000 flag 1
            end

            // Claim Priority Lock 3 and arm 3000ms release watchdog
            0085: 30@ = 31@
            000A: 30@ += MEM_PRIORITY_LOCK
            0A8C: write_memory 30@ size 4 value 3 virtual_protect 0
            13@ = 3000
        end

        // Rematerialize Shion beside Jo'on on floor level
        04C4: store_coords_to 20@ 21@ 22@ from_actor $PLAYER_ACTOR with_offset 0.0 -1.2 0.0
        000F: 22@ -= 1.0
        00A1: put_actor 10@ at 20@ 21@ 22@

        077E: get_active_interior_to 28@
        0860: link_actor 10@ to_interior 28@
        0D10: set_actor 10@ alpha 255
        0619: enable_actor 10@ collision_detection 1

        // Face Jo'on
        0639: AS_actor 10@ rotate_to_actor $PLAYER_ACTOR

        // Ethereal Blue Hitodama Arrival Flare on Shion
        0A92: create_custom_thread "Shion\FX_Warp.c" 1 10@ 0.0 0.0

        // Re-link to group and release busy shield
        07AF: 20@ = player $PLAYER_CHAR group
        0631: put_actor 10@ in_group 20@
        0085: 30@ = 31@
        000A: 30@ += MEM_SHION_BUSY
        0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

        1@ = 0
    end
end

goto @ShionBurglary_MainLoop


// =========================================================================
// SUBROUTINES
// =========================================================================

// --- EXECUTE THE CO-OP HAND-OFF ---
:Execute_Loot_Handoff
// Wipe carrying task & secondary upper-body blend animation from Jo'on
0E97: clear_char_secondary_tasks $PLAYER_ACTOR
0E96: clear_char_primary_tasks $PLAYER_ACTOR
0792: clear_char_tasks_immediately $PLAYER_ACTOR
0687: clear_actor $PLAYER_ACTOR task

// Clear BURGJB mission memory so BURGJB immediately unlocks the next item!
0A8C: write_memory 0xA48AD8 size 4 value 0 virtual_protect 0
0A8C: write_memory 0xA48A24 size 4 value 0 virtual_protect 0
0A8C: write_memory 0xA48AF0 size 4 value 0 virtual_protect 0

// Detach and relocate object far underground
if
    0019: 11@ > 0
then
    if
        03CA: object 11@ exists
    then
        0682: detach_object 11@ 0.0 0.0 0.0 collision_detection 0
        0382: set_object 11@ collision_detection 0
        01BC: put_object 11@ at 0.0 0.0 -500.0
    end
end

// Reset return lock timer if a new hand-off begins
13@ = 0

// Voice Lines: 40% chance to fire (Claim Tier 3 Action Priority)
0085: 30@ = 31@
000A: 30@ += MEM_PRIORITY_LOCK
0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
if
    8029: not 19@ >= 4 // Do not stomp on Tier 4 Theater
then
    0209: 25@ = random_int_in_ranges 0 100
    if
        8019: not 25@ > 40
    then
        // Preempt all ambient banter so Shion's handoff has exclusive focus
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Sub"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Int"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Zone"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Mny"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Idle"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Snip"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Veh"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Rad"
        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Wthr"

        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8C: write_memory 30@ size 4 value 3 virtual_protect 0
        018C: play_sound 1058 at 0.0 0.0 0.0
        03EB: clear_small_messages_only

        0209: 26@ = random_int_in_ranges 0 3
        if
            003B: 26@ == 3@
        then
            0209: 27@ = random_int_in_ranges 1 3
            005A: 26@ += 27@
            if
                0029: 26@ >= 3
            then
                000E: 26@ -= 3
            end
        end
        0085: 3@ = 26@

        if
            0039: 26@ == 0
        then
            0ADF: add_text_label "Y_BGHF1" text "~b~Shion: ~s~\"Ugh, it's heavy... taking this to the van now. You're buying dinner for this, Jo'on.\""
            09C1: add_next_text_to_brief_history 1
            00BC: show_text_highpriority GXT 'Y_BGHF1' time 3000 flag 1
        end
        if
            0039: 26@ == 1
        then
            0ADF: add_text_label "Y_BGHF2" text "~b~Shion: ~s~\"Taking it outside... my spine's already killing me. Don't make me carry twenty more of these.\""
            09C1: add_next_text_to_brief_history 1
            00BC: show_text_highpriority GXT 'Y_BGHF2' time 3000 flag 1
        end
        if
            0039: 26@ == 2
        then
            0ADF: add_text_label "Y_BGHF3" text "~b~Shion: ~s~\"Fine, I'll drag it to the van... don't make so much noise while I'm gone or we're dead.\""
            09C1: add_next_text_to_brief_history 1
            00BC: show_text_highpriority GXT 'Y_BGHF3' time 3000 flag 1
        end
    else
        // Silent handoff audio feedback
        018C: play_sound 1058 at 0.0 0.0 0.0
    end
end

// Raise Busy Shield so SpawnShion steps aside
0085: 30@ = 31@
000A: 30@ += MEM_SHION_BUSY
0A8C: write_memory 30@ size 4 value 1 virtual_protect 0

// Departure Hitodama burst at Shion's position
00A0: store_actor 10@ position_to 20@ 21@ 22@
0A92: create_custom_thread "Shion\FX_Warp.c" 0 20@ 21@ 22@

// Remove from group, hide Shion, and disable collision
07AF: 20@ = player $PLAYER_CHAR group
06C9: remove_actor 10@ from_group
0792: clear_char_tasks_immediately 10@
0D10: set_actor 10@ alpha 0
0619: enable_actor 10@ collision_detection 0

2@ = 0
1@ = 1 // Switch to Van Spirit Run State
return

// --- RESTORE SHION ON EMERGENCY / CANCEL ---
:Restore_Shion_Burglary_State
if
    0019: 11@ > 0
then
    if
        03CA: object 11@ exists
    then
        0682: detach_object 11@ 0.0 0.0 0.0 collision_detection 0
        0382: set_object 11@ collision_detection 0
        01BC: put_object 11@ at 0.0 0.0 -500.0
    end
    11@ = 0
end

if and
    0019: 10@ > 0
    056D: actor 10@ defined
    8118: not actor 10@ dead
then
    // If caught mid-run, warp back to Jo'on
    if
        0039: 1@ == 1
    then
        04C4: store_coords_to 20@ 21@ 22@ from_actor $PLAYER_ACTOR with_offset 0.0 -1.2 0.0
        000F: 22@ -= 1.0
        00A1: put_actor 10@ at 20@ 21@ 22@
        077E: get_active_interior_to 28@
        0860: link_actor 10@ to_interior 28@
    end

    0792: clear_char_tasks_immediately 10@
    0D10: set_actor 10@ alpha 255
    0619: enable_actor 10@ collision_detection 1
    07AF: 20@ = player $PLAYER_CHAR group
    0631: put_actor 10@ in_group 20@
end

0085: 30@ = 31@
000A: 30@ += MEM_SHION_BUSY
0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

0085: 30@ = 31@
000A: 30@ += MEM_PRIORITY_LOCK
0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
if
    0039: 19@ == 3
then
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
end

1@ = 0
2@ = 0
12@ = 0
13@ = 0
return
