{$CLEO .c}
{$USE CLEO+}
{$USE memory}
{$USE ini}

0000: NOP
script_name "Sh_Sync"

// =========================================================================
// YORIGAMI MASTER HEAP DICTIONARY (PRUNED FOR SH_SYNC)
// =========================================================================
const
    MEM_SHION_BUSY = 4            // Formerly 708
    MEM_PRIORITY_LOCK = 20        // Formerly 765
end

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
// 31@ NOW HOLDS THE MASTER POINTER.

// 0@ = Shion Actor Handle passed from SpawnShion
0085: 10@ = 0@

13@ = 0 // Riding Together Confirmed Latch (0 = Apart, 1 = Riding Together)
14@ = 0 // Single-shot exit command latch

// Read ModifierKey once on boot into dedicated register 2@
0AF0: 2@ = read_int_from_ini_file "cleo\Shion.ini" section "Controls" key "ModifierKey"
if
    2@ <= 0
then
    2@ = 89 // Default: Y Key
end

:ShionVehicle_Loop
wait 50

if or
    8256: not is_player_playing $PLAYER_CHAR
    10@ <= 0
    856D: not actor 10@ defined
    0118: actor 10@ dead
then
    0A93: terminate_this_custom_script
end

// =========================================================================
// IN-VEHICLE WARP ENGINE (Manual [G] OR Automatic Distance > 45m)
// =========================================================================
if and
    00DF: actor $PLAYER_ACTOR driving
    816B: not fading
    8E25: not is_on_cutscene
then
    03C0: 24@ = actor $PLAYER_ACTOR car
    if and
        056E: car 24@ defined
        8119: not car 24@ wrecked
        80DB: not actor 10@ in_car 24@ // Shion is NOT in player's car
        8449: not actor 10@ in_a_car   // Shion is on foot outside
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_SHION_BUSY
        0A8D: 11@ = read_memory 30@ size 4 virtual_protect 0
        if
            11@ == 0 // Not actively digging in trash or skydiving
        then
            23@ = 0 // 1 = Trigger Warp Sequence (Integer flag)
            25@ = 0 // 0 = Auto Distance Warp (Triggers Banter), 1 = Manual [G] (Silent/No Banter)

            // Trigger A: Manual [G] Keypress (Intentional)
            if and
                0AB0: is_key_pressed 71 // G Key (VK 71)
                8AB0: not is_key_pressed 2@ // ModifierKey (Y) NOT held
            then
                :Wait_Vehicle_G_Rel
                wait 50
                if
                    0AB0: is_key_pressed 71
                then
                    goto @Wait_Vehicle_G_Rel
                end
                23@ = 1
                25@ = 1 // Flag as manual command
            end

            // Trigger B: Automatic Distance Check (> 45.0m - Ditched!)
            if and
                23@ == 0
                8818: not actor 10@ in_air // Not jumping/falling
            then
                00A0: store_actor $PLAYER_ACTOR position_to 20@ 21@ 22@
                00A0: store_actor 10@ position_to 17@ 18@ 19@
                050A: 7@ = distance_between_XYZ 20@ 21@ 22@ and_XYZ 17@ 18@ 19@
                if
                    0021: 7@ > 45.0
                then
                    23@ = 1
                    25@ = 0 // Flag as auto-catchup (Triggers banter)
                end
            end

            // Execute In-Vehicle Warp
            if
                23@ == 1
            then
                01EA: 5@ = car 24@ max_passengers
                if
                    5@ <= 0
                then
                    // Single-seater vehicle (Tank, Jet, Kart, etc.)
                    if
                        25@ == 1 // Only notify if player pressed [G] manually
                    then
                        018C: play_sound 1057 at 0.0 0.0 0.0
                        03EB: clear_small_messages_only
                        0AD1: print_formatted_now "~r~No passenger seats available in this vehicle!" 2500
                    end
                else
                    // Check Seat 0 (Front passenger / Bike pillion)
                    28@ = -1
                    if
                        0431: car 24@ passenger_seat_free 0
                    then
                        28@ = 0
                    else
                        // If Seat 0 is occupied, check Seat 1 (Rear Left)
                        if
                            5@ > 1
                        then
                            if
                                0431: car 24@ passenger_seat_free 1
                            then
                                28@ = 1
                            else
                                // If Seat 1 is occupied, check Seat 2 (Rear Right)
                                if
                                    5@ > 2
                                then
                                    if
                                        0431: car 24@ passenger_seat_free 2
                                    then
                                        28@ = 2
                                    end
                                end
                            end
                        end
                    end

                    if
                        28@ >= 0
                    then
                        // Instantly terminate all running chat scripts and clear prints
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Veh"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Rad"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Zone"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Mny"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Idle"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Snip"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Int"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Sub"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Star"
                        0ABA: terminate_all_custom_scripts_with_this_name "Sh_Wthr"
                        03EB: clear_small_messages_only

                        // Departure FX (Static burst left behind on pavement)
                        00A0: store_actor 10@ position_to 17@ 18@ 19@
                        0A92: create_custom_thread "Shion\FX_Warp.c" 0 17@ 18@ 19@

                        // Warp Shion into seat immediately
                        0792: clear_char_tasks_immediately 10@
                        warp_char_into_car_as_passenger 10@ into_car 24@ passenger_seat 28@
                        task_enter_car_as_passenger 10@ enter_car 24@ passenger_seat 28@ time 5000

                        // Arrival FX (Mode 1: Glued to Shion inside moving vehicle)
                        0A92: create_custom_thread "Shion\FX_Warp.c" 1 10@ 0.0 0.0

                        13@ = 1 // Latch riding together
                        14@ = 0 // Clear exit latch

                        018C: play_sound 1058 at 0.0 0.0 0.0
                        
                        // Play banter only on unprompted auto-warp
                        if
                            25@ == 0 // Auto distance warp
                        then
                            0085: 30@ = 31@
                            000A: 30@ += MEM_PRIORITY_LOCK
                            0A8D: 9@ = read_memory 30@ size 4 virtual_protect 0
                            if
                                9@ <= 3 // Do not downgrade higher priority locks
                            then
                                0A8C: write_memory 30@ size 4 value 3 virtual_protect 0 // Tier 3 Action Lock
                                gosub @Play_Warp_Catchup_Dialogue
                            end
                        end
                    else
                        if
                            25@ == 1 // Only notify if player pressed [G] manually
                        then
                            018C: play_sound 1057 at 0.0 0.0 0.0
                            03EB: clear_small_messages_only
                            0AD1: print_formatted_now "~r~All passenger seats are currently occupied!" 2500
                        end
                    end
                end
            end
        end
    end
end

// =========================================================================
// PASSENGER EXIT & AIRBORNE BAILOUT ENGINE
// =========================================================================
if
    0449: actor 10@ in_a_car
then
    03C0: 27@ = actor 10@ car
    if
        056E: car 27@ defined
    then
        // Latch that Jo'on and Shion are riding together inside this vehicle
        if
            00DB: actor $PLAYER_ACTOR in_car 27@
        then
            13@ = 1
        end

        8@ = 0 // 1 = Shion must disembark immediately

        // Emergency! Car caught on fire
        0227: 29@ = car 27@ health
        if
            29@ < 250
        then
            8@ = 1
        end

        // They were riding together, but Jo'on is now outside on foot
        if and
            8@ == 0
            13@ == 1
            80DB: not actor $PLAYER_ACTOR in_car 27@
        then
            8@ = 1
        end

        // They were riding together, and Jo'on just initiated vehicle exit
        if and
            8@ == 0
            13@ == 1
            0E4A: is_char_exiting_any_car $PLAYER_ACTOR
        then
            8@ = 1
        end

        // Jo'on entered or hijacked a different vehicle
        if and
            8@ == 0
            00DF: actor $PLAYER_ACTOR driving
        then
            03C0: 24@ = actor $PLAYER_ACTOR car
            if
                803B: not 27@ == 24@
            then
                8@ = 1
            end
        end

        // Execute Immediate Exit / Bailout
        if and
            8@ == 1
            14@ == 0
        then
            14@ = 1 // Lock command
            13@ = 0 // Clear riding latch

            // Measure true height above terrain using isolated float register 16@
            00AA: store_car 27@ position_to 20@ 21@ 22@
            02CE: 16@ = ground_z_at 20@ 21@ 22@
            0063: 22@ -= 16@

            // Check if vehicle is airborne and high up
            if and
                01F3: car 27@ in_air
                0021: 22@ > 12.0
            then
                // Aircraft Ejection into freefall
                0407: store_coords_to 20@ 21@ 22@ from_car 27@ with_offset 0.0 -2.5 -1.2
                0362: remove_actor 10@ from_car_and_place_at 20@ 21@ 22@
                083C: set_actor 10@ velocity_in_direction_XYZ 0.0 0.0 0.5
            else
                // Ground Vehicle Check
                02E3: 6@ = car 27@ speed
                if
                    0021: 6@ > 10.0 // Car rolling fast
                then
                    0622: AS_actor 10@ bail_car 27@
                else
                    0633: AS_actor 10@ exit_car
                end
            end
        end
    end
else
    // Shion is outside on foot: Reset latches
    14@ = 0
    13@ = 0
end

goto @ShionVehicle_Loop

    03EB: clear_small_messages_only

    09C1: add_next_text_to_brief_history 1
    00BC: show_text_highpriority GXT 'Y_CW1B' time 3000 flag 1
