    010B: 14@ = player $PLAYER_CHAR money

            22@ = 1

if and

    25@ == 0
    9@ <= 0  // Smoke dialogue off cooldown

    000A: 30@ += MEM_SHION_HANDLE

            29@ = 1
        end
    end

if
    29@ == 1

    0209: 14@ = random_int_in_ranges 0 100
    if

    then
        29@ = 1
        0209: 26@ = random_int_in_ranges 0 3

if or
    8256: not is_player_playing $PLAYER_CHAR
    0118: actor $PLAYER_ACTOR dead
    02E0: actor $PLAYER_ACTOR firing_weapon
    0E67: is_aim_button_pressed 0
    0E8D: is_any_fire_button_pressed 0
    0E47: is_char_fighting $PLAYER_ACTOR

if or
    00E1: player 0 pressed_key 17 // Fire Key
    00E1: player 0 pressed_key 6  // Aim Key
    0E49: is_char_entering_any_car $PLAYER_ACTOR
    0E4A: is_char_exiting_any_car $PLAYER_ACTOR

    016B: fading
    0E25: is_on_cutscene

    13@ = 1
end

// Abort active smoking immediately if burglary mission is active

    0038: $ONMISSION_BURGLARY == 1
then
    13@ = 1
end

// Abort active smoking immediately if an incoming phone call is picked up
if or
    0E42: is_char_doing_task_id $PLAYER_ACTOR 1600
    0E42: is_char_doing_task_id $PLAYER_ACTOR 1601
    0E42: is_char_doing_task_id $PLAYER_ACTOR 1602
    0E42: is_char_doing_task_id $PLAYER_ACTOR 1603
then
    13@ = 1
end

if or
    0611: actor $PLAYER_ACTOR performing_animation "phone_talk"
    0611: actor $PLAYER_ACTOR performing_animation "phone_in"
    0611: actor $PLAYER_ACTOR performing_animation "phone_out"
then
    13@ = 1
end
// Vehicle state abort checks while inside a car
if
    0449: actor $PLAYER_ACTOR in_a_car
then
    if
        047A: actor $PLAYER_ACTOR driving_bike
    then
        13@ = 1
    else
        03C0: 25@ = actor $PLAYER_ACTOR car
        if or
            0119: car 25@ wrecked
            01F4: car 25@ flipped
            02BF: car 25@ sunk
        then
            13@ = 1
        end

        02E3: 20@ = car 25@ speed
        if
            0021: 20@ > 3.0 // Accelerating or driving immediately aborts smoking
        then
            13@ = 1
        end
    end

    // On-foot abort checks
    06AC: 20@ = actor $PLAYER_ACTOR movement_speed
    if
        0021: 20@ > 2.0 // Jogging or running
    then
        13@ = 1
    end

    if and
        00E1: player 0 pressed_key 16 // Sprinting
        0021: 20@ > 0.5
    then
        13@ = 1
    end

    if
        0818: actor $PLAYER_ACTOR in_air // Jumping or falling
    then
        13@ = 1
    end
end

// Abort if smoking animation was cancelled on foot by engine tasks
if
    8449: not actor $PLAYER_ACTOR in_a_car
then
    if
        TIMERA >= 500
    then
        if
            8611: not actor $PLAYER_ACTOR performing_animation "SMKCIG_PRTL"
        then
            13@ = 1
        end
    end
end

// Manual toggle off (F11) - Debounced after 500ms to prevent double-tap cancellation

        0AB0: is_key_pressed 122
    then
        :Wait_For_Cancel_F11

        if
            0AB0: is_key_pressed 122
        then
            goto @Wait_For_Cancel_F11
        end
        13@ = 1
    end
end

// Also abort if player is no longer Jo'on
0085: 30@ = 31@
000A: 30@ += MEM_PLAYER_FORM
0A8D: 11@ = read_memory 30@ size 4 virtual_protect 0
if
    8039: not 11@ == 1
then
    13@ = 1
end

if
    13@ == 1
then
    goto @Cleanup_Smoking_State
end

// Dynamic preemption check: Surrender mic if higher tier claims channel or mission starts

    29@ == 1
then
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8D: 14@ = read_memory 30@ size 4 virtual_protect 0
    if
        14@ > 1
    then
        29@ = 0
    else
        if or
            0038: $ONMISSION == 1
            0E1D: is_on_mission
        then
            03EB: clear_small_messages_only
            if
                14@ == 1
            then
                0085: 30@ = 31@
                000A: 30@ += MEM_PRIORITY_LOCK
                0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

            29@ = 0
        end
    end

// Non-blocking Shion commentary stages

    if and
        28@ == 1

        gosub @Dispatch_Joon_Smoke_Line_B

    3@ == 0
    TIMERA >= 4200

    0669: 2@ = attach_particle "EXHALE" to_actor $PLAYER_ACTOR with_offset 0.0 0.12 0.0 type 0
    0883: attach_particle 2@ to_actor $PLAYER_ACTOR mode 5
    064C: make_particle 2@ visible

    TIMERA >= 6500

// =========================================================================

// Destroy primary cigar prop handle

    1@ > 0

    if
        03CA: object 1@ exists
    then
        0682: detach_object 1@ 0.0 0.0 0.0 collision_detection 0
        0108: destroy_object 1@
    end
    1@ = 0
end
0085: 30@ = 31@
000A: 30@ += MEM_JOON_CIGAR_OBJ
0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

// Sweep vicinity for any detached orphan cigar models using isolated scratch registers (10@ 11@ 13@)

    056D: actor $PLAYER_ACTOR defined
    8118: not actor $PLAYER_ACTOR dead
then
    00A0: store_actor $PLAYER_ACTOR position_to 10@ 11@ 13@
    24@ = 0

    :Cigar_Cleanup_Purge
    if
        0AE3: 25@ = random_object_near_point 10@ 11@ 13@ in_radius 4.0 find_next 24@
    then
        24@ = 1
        0984: 23@ = object 25@ model
        if
            23@ == #CIGAR
        then
            0682: detach_object 25@ 0.0 0.0 0.0 collision_detection 0
            0108: destroy_object 25@
        end
        goto @Cigar_Cleanup_Purge
    end
end

// Release smoke particle reference so it lingers and disperses in the air

    2@ > 0
then
    064F: remove_references_to_particle 2@
    2@ = 0
end

// Immediately restore previous weapon

    4@ > 0
then
    01B9: set_actor $PLAYER_ACTOR armed_weapon_to 4@
end

// Release Tier 1 Priority lock if held

    29@ == 1
then
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8D: 14@ = read_memory 30@ size 4 virtual_protect 0
    if
        14@ == 1
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
    end
end

0085: 7@ = 6@ // Set Cooldown Timer
8@ = 0        // Reset Idle Hands Timer
0@ = 0        // Clear Active Flag
3@ = 0        // Clear Smoke Puff Latch
28@ = 0       // Clear Dialogue Step
29@ = 0       // Clear Dialogue Active Flag
