{$CLEO .c}
{$USE CLEO+}
{$USE memory}

0000: NOP
script_name "Sh_TrkM"

wait 1200

// =========================================================================
// YORIGAMI MASTER HEAP DICTIONARY (PRUNED FOR SHION TRACK MISSIONS)
// =========================================================================
const
    // --- SHION ENTITY & INVENTORY DATA ---
    MEM_SHION_HANDLE = 116

    // --- SHIONMEMORIES.INI TELEMETRY (THE LEDGER) ---
    MEM_PIMPING_LEVEL = 240
    MEM_BURGLARIES = 264
    MEM_PARAMEDIC_SAVES = 280
    MEM_FIREFIGHT_EXTING = 260
    MEM_VIGILANTE_KILLS = 352
    MEM_TAXI_FARES = 356
end

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
// 31@ NOW HOLDS THE MASTER POINTER.

9@ = 0  // Burglary Fenced State Latch (0 = Ready, 1 = Latched)
14@ = 0 // Tracked Paramedic Patients Saved (Stat 151)
15@ = 0 // Tracked Pimping Mission Level (0xA48A24)
16@ = 0 // Tracked Fires Extinguished (Stat 153)
17@ = 0 // Tracked Vigilante Kills (Stat 152)
18@ = 0 // Tracked Taxi Fares (Stat 150)

// Initialize baselines from current savegame
0652: 14@ = integer_stat 151 // People saved in an Ambulance
0652: 16@ = integer_stat 153 // Total fires extinguished
0652: 17@ = integer_stat 152 // Criminals killed on Vigilante Mission
0652: 18@ = integer_stat 150 // Passengers dropped off

:ShionTrackMissions_Loop
wait 250

if
    8256: not player $PLAYER_CHAR defined
then
    goto @ShionTrackMissions_Loop
end

// Read Live Shion Handle from Shared Memory
0085: 30@ = 31@
000A: 30@ += MEM_SHION_HANDLE
0A8D: 10@ = read_memory 30@ size 4 virtual_protect 0

// Bailout and sync baselines if Shion is dismissed (0) or dead in solid form
if
    10@ == 0
then
    9@ = 0
    15@ = 0
    0652: 14@ = integer_stat 151
    0652: 16@ = integer_stat 153
    0652: 17@ = integer_stat 152
    0652: 18@ = integer_stat 150
    goto @ShionTrackMissions_Loop
end

if
    10@ > 0
then
    if or
        856D: not actor 10@ defined
        0118: actor 10@ dead
    then
        9@ = 0
        15@ = 0
        0652: 14@ = integer_stat 151
        0652: 16@ = integer_stat 153
        0652: 17@ = integer_stat 152
        0652: 18@ = integer_stat 150
        goto @ShionTrackMissions_Loop
    end
end

// =========================================================================
// PIMPING WITH SHION TRACKER (Hitodama & Physical)
// Tracks successful client drop-offs during Pimping sub-mission with Shion
// =========================================================================
if and
    0038: $ONMISSION == 1
    0038: $162 == 1
then
    0A8D: 28@ = read_memory 0xA48A24 size 4 virtual_protect 0
    if
        001D: 28@ > 15@
    then
        0085: 15@ = 28@

        if
            0449: actor $PLAYER_ACTOR in_a_car
        then
            03C0: 27@ = actor $PLAYER_ACTOR car
            23@ = 0
            if
                10@ == -1
            then
                23@ = 1
            else
                if
                    00DB: actor 10@ in_car 27@
                then
                    23@ = 1
                end
            end

            if
                23@ == 1
            then
                0085: 30@ = 31@
                000A: 30@ += MEM_PIMPING_LEVEL
                0A8D: 25@ = read_memory 30@ size 4 virtual_protect 0
                000A: 25@ += 1
                0085: 30@ = 31@
                000A: 30@ += MEM_PIMPING_LEVEL
                0A8C: write_memory 30@ size 4 value 25@ virtual_protect 0
            end
        end
    end
else
    15@ = 0
end

// =========================================================================
// BURGLARY FENCED WATCHDOG (Hitodama & Physical - Latched with 9@)
// Tracks successful lockup garage drop-offs during Burglary with Shion
// =========================================================================
if
    0038: $ONMISSION_BURGLARY == 1
then
    00A0: store_actor $PLAYER_ACTOR position_to 24@ 25@ 26@
    28@ = 0

    // LS Lockup (Playa del Seville)
    050A: 27@ = distance_between_XYZ 24@ 25@ 26@ and_XYZ 2738.0 -2013.0 13.0
    if
        8021: not 27@ > 10.0
    then
        28@ = 1
    end

    // SF Lockup (Doherty)
    050A: 27@ = distance_between_XYZ 24@ 25@ 26@ and_XYZ -2102.5 -15.7 35.3
    if
        8021: not 27@ > 10.0
    then
        28@ = 1
    end

    // LV Lockup (Pilgrim / Camel's Toe)
    050A: 27@ = distance_between_XYZ 24@ 25@ 26@ and_XYZ 2609.3 1437.9 10.8
    if
        8021: not 27@ > 10.0
    then
        28@ = 1
    end

    if
        28@ == 1
    then
        if
            9@ == 0
        then
            if
                00DF: actor $PLAYER_ACTOR driving
            then
                03C0: 27@ = actor $PLAYER_ACTOR car
                if
                    01C1: car 27@ stopped
                then
                    0441: 26@ = car 27@ model
                    if
                        26@ == 609 // Black Boxville
                    then
                        23@ = 0
                        if
                            10@ == -1
                        then
                            23@ = 1
                        else
                            if
                                00DB: actor 10@ in_car 27@
                            then
                                23@ = 1
                            end
                        end

                        if
                            23@ == 1
                        then
                            0085: 30@ = 31@
                            000A: 30@ += MEM_BURGLARIES
                            0A8D: 25@ = read_memory 30@ size 4 virtual_protect 0
                            000A: 25@ += 1
                            0085: 30@ = 31@
                            000A: 30@ += MEM_BURGLARIES
                            0A8C: write_memory 30@ size 4 value 25@ virtual_protect 0
                            9@ = 1
                        end
                    end
                end
            end
        end
    else
        9@ = 0
    end
else
    9@ = 0
end

// =========================================================================
// PARAMEDIC WITH SHION TRACKER (Stat 151: People saved in an Ambulance)
// =========================================================================
0652: 28@ = integer_stat 151
if
    001D: 28@ > 14@
then
    0085: 29@ = 28@
    0062: 29@ -= 14@
    0085: 14@ = 28@

    23@ = 0
    if
        10@ == -1
    then
        23@ = 1
    else
        if
            0449: actor $PLAYER_ACTOR in_a_car
        then
            03C0: 27@ = actor $PLAYER_ACTOR car
            if
                00DB: actor 10@ in_car 27@
            then
                23@ = 1
            end
        end
    end

    if
        23@ == 1
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_PARAMEDIC_SAVES
        0A8D: 25@ = read_memory 30@ size 4 virtual_protect 0
        005A: 25@ += 29@
        0085: 30@ = 31@
        000A: 30@ += MEM_PARAMEDIC_SAVES
        0A8C: write_memory 30@ size 4 value 25@ virtual_protect 0
    end
else
    0085: 14@ = 28@
end

// =========================================================================
// FIREFIGHTER WITH SHION TRACKER (Stat 153: Total fires extinguished)
// =========================================================================
0652: 28@ = integer_stat 153
if
    001D: 28@ > 16@
then
    0085: 29@ = 28@
    0062: 29@ -= 16@
    0085: 16@ = 28@

    23@ = 0
    if
        10@ == -1
    then
        23@ = 1
    else
        if
            0449: actor $PLAYER_ACTOR in_a_car
        then
            03C0: 27@ = actor $PLAYER_ACTOR car
            if
                00DB: actor 10@ in_car 27@
            then
                23@ = 1
            end
        end
    end

    if
        23@ == 1
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_FIREFIGHT_EXTING
        0A8D: 25@ = read_memory 30@ size 4 virtual_protect 0
        005A: 25@ += 29@
        0085: 30@ = 31@
        000A: 30@ += MEM_FIREFIGHT_EXTING
        0A8C: write_memory 30@ size 4 value 25@ virtual_protect 0
    end
else
    0085: 16@ = 28@
end

// =========================================================================
// VIGILANTE WITH SHION TRACKER (Stat 152: Criminals killed on Vigilante)
// =========================================================================
0652: 28@ = integer_stat 152
if
    001D: 28@ > 17@
then
    0085: 29@ = 28@
    0062: 29@ -= 17@
    0085: 17@ = 28@

    23@ = 0
    if
        10@ == -1
    then
        23@ = 1
    else
        if
            0EE4: locate_char_distance_to_char $PLAYER_ACTOR char 10@ radius 50.0
        then
            23@ = 1
        end
    end

    if
        23@ == 1
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_VIGILANTE_KILLS
        0A8D: 25@ = read_memory 30@ size 4 virtual_protect 0
        005A: 25@ += 29@
        0085: 30@ = 31@
        000A: 30@ += MEM_VIGILANTE_KILLS
        0A8C: write_memory 30@ size 4 value 25@ virtual_protect 0
    end

    0085: 17@ = 28@

// =========================================================================
// TAXI FARES WITH SHION TRACKER (Stat 150: Passengers dropped off)
// =========================================================================
0652: 28@ = integer_stat 150
if
    001D: 28@ > 18@
then
    0085: 29@ = 28@
    0062: 29@ -= 18@
    0085: 18@ = 28@

    23@ = 0
    if
        10@ == -1
    then
        23@ = 1
    else
        if
            0449: actor $PLAYER_ACTOR in_a_car
        then
            03C0: 27@ = actor $PLAYER_ACTOR car
            if
                00DB: actor 10@ in_car 27@
            then

    end

    if
        23@ == 1
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_TAXI_FARES
        0A8D: 25@ = read_memory 30@ size 4 virtual_protect 0
        005A: 25@ += 29@
        0085: 30@ = 31@
        000A: 30@ += MEM_TAXI_FARES
        0A8C: write_memory 30@ size 4 value 25@ virtual_protect 0
    end
else
    0085: 18@ = 28@
end

goto @ShionTrackMissions_Loop
