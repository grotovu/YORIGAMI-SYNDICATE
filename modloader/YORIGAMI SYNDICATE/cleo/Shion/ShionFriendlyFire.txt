{$CLEO .c}
{$USE CLEO+}
{$USE memory}

0000: NOP

// =========================================================================
// YORIGAMI HEAP CONSTANTS (FILTERED FOR SHION_FRIENDLYFIRE)
// =========================================================================
const
    MEM_PLAYER_FORM   = 12
    MEM_PRIORITY_LOCK = 20
    MEM_SHION_HANDLE  = 116
    MEM_HITODAMA_X    = 144
    MEM_HITODAMA_Y    = 148
    MEM_HITODAMA_Z    = 152
end

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
// 31@ NOW HOLDS THE MASTER POINTER.

script_name "Sh_FF"

// 0@ = Shion Actor Handle passed from SpawnShion
0085: 10@ = 0@

1@ = 0  // Debounce / Cooldown Timer (ms)
11@ = 0 // Pending Friendly Fire Latch (0 = None, 1 = Triggered)
12@ = 0 // Damage Weapon Type
13@ = 0 // Tracked Camera Ammo Baseline

// Register Damage Callback specifically for Friendly Fire
0EDE: set_script_event_char_damage 1 @OnCharDamageFF 19@

:ShionFF_Loop
wait 50

if
    8256: not is_player_playing $PLAYER_CHAR
then
    goto @ShionFF_Bailout
end

// Read Live Shion State (0 = Inactive/Lost, -1 = Hitodama Spirit, >0 = Physical Ped)
0085: 30@ = 31@
000A: 30@ += MEM_SHION_HANDLE
0A8D: 18@ = read_memory 30@ size 4 virtual_protect 0
if
    18@ == 0
then
    goto @ShionFF_Bailout
end

// Physical Ped Validity Check
if
    18@ > 0
then
    0085: 10@ = 18@
    if or
        856D: not actor 10@ defined
        0118: actor 10@ dead
    then
        goto @ShionFF_Bailout
    end
else
    10@ = -1
end

// Tick down debounce / cooldown timer
if
    1@ > 0
then
    000E: 1@ -= 50
    if
        1@ <= 0
    then
        1@ = 0
    end
end

// =========================================================================
// CAMERA DETECTION: NATIVE PED PHOTOGRAPHY & HITODAMA OCCULT SCANNER
// =========================================================================
// Always track live camera ammo every tick to prevent stale baseline re-triggers
0470: 24@ = actor $PLAYER_ACTOR current_weapon
if
    24@ == 43
then
    041A: 25@ = actor $PLAYER_ACTOR weapon 43 ammo

    // Sync baseline on first weapon draw
    if
        13@ <= 0
    then
        0085: 13@ = 25@
    end

    27@ = 0 // Photo Taken Flag

    // Check if camera ammo dropped by at least 1
    if and
        13@ > 0
        001D: 13@ > 25@
    then
        27@ = 1
    end

    // Always keep tracked baseline synced to live ammo every frame
    0085: 13@ = 25@

    // Backup check for infinite ammo: single-frame shutter click pulse while aiming
    if and
        27@ == 0
        1@ <= 0
    then
        if or
            0E67: is_aim_button_pressed 0
            00E1: player 0 pressed_key 6
        then
            if or
                0E3D: is_key_just_pressed 1 // Left Mouse Button click
                0E3E: is_button_just_pressed 0 button 17 // Controller fire button pulse
            then
                27@ = 1
            end
        end
    end
else
    13@ = 0 // Reset baseline when camera is put away
    27@ = 0
end

// Evaluate photo target only when debounce is off cooldown and playing as Jo'on
if
    1@ <= 0
then
    0085: 30@ = 31@
    000A: 30@ += MEM_PLAYER_FORM
    0A8D: 17@ = read_memory 30@ size 4 virtual_protect 0
    if
        17@ == 1
    then
        // Physical Ped Photography (Native Opcode 04C5)
        if
            10@ > 0
        then
            if and
                056D: actor 10@ defined
                8118: not actor 10@ dead
            then
                if
                    04C5: actor 10@ photographed
                then
                    0085: 30@ = 31@
                    000A: 30@ += MEM_PRIORITY_LOCK
                    0A8D: 16@ = read_memory 30@ size 4 virtual_protect 0
                    if
                        16@ < 4
                    then
                        11@ = 1
                        12@ = 43
                    end
                end
            end
        else
            // Hitodama Spirit Photography (Frustum & Viewfinder Detection)
            if and
                10@ == -1
                27@ == 1 // Photo was actually taken this tick
            then
                0085: 30@ = 31@
                000A: 30@ += MEM_HITODAMA_X
                0A8D: 20@ = read_memory 30@ size 4 virtual_protect 0

                0085: 30@ = 31@
                000A: 30@ += MEM_HITODAMA_Y
                0A8D: 21@ = read_memory 30@ size 4 virtual_protect 0

                0085: 30@ = 31@
                000A: 30@ += MEM_HITODAMA_Z
                0A8D: 22@ = read_memory 30@ size 4 virtual_protect 0

                if or
                    8043: not 20@ == 0.0
                    8043: not 21@ == 0.0
                then
                    00A0: store_actor $PLAYER_ACTOR position_to 7@ 8@ 9@
                    050A: 15@ = distance_between_XYZ 20@ 21@ 22@ and_XYZ 7@ 8@ 9@
                    if
                        8021: not 15@ > 35.0
                    then
                        if
                            00C2: sphere_onscreen 20@ 21@ 22@ radius 2.5
                        then
                            0085: 30@ = 31@
                            000A: 30@ += MEM_PRIORITY_LOCK
                            0A8D: 16@ = read_memory 30@ size 4 virtual_protect 0
                            if
                                16@ < 4
                            then
                                11@ = 1
                                12@ = 443
                            end
                        end
                    end
                end
            end
        end
    end
end

// Process Pending Friendly Fire Trigger
if
    11@ == 1
then
    11@ = 0
    1@ = 3500 // Engage debounce to prevent rapid-hit spam
    0A92: stream_custom_script "chatFriendlyFire.c" 12@
end

goto @ShionFF_Loop


// =========================================================================
// DAMAGE EVENT CALLBACK
// =========================================================================
:OnCharDamageFF
if and
    056D: actor 19@ defined
    8118: not actor 19@ dead
then
    if
        10@ > 0
    then
        if
            003B: 19@ == 10@
        then
            // Only detect friendly fire on foot (ignores firetruck hose and vehicle bumps)
            if
                80DF: not actor $PLAYER_ACTOR driving
            then
                // Guard: Only Jo'on triggers sisterly friendly-fire banter
                0085: 14@ = 31@
                000A: 14@ += MEM_PLAYER_FORM
                0A8D: 7@ = read_memory 14@ size 4 virtual_protect 0
                if
                    7@ == 1
                then
                    // 2@ = Damager Struct, 3@ = Weapon Type, 4@ = Body Part, 5@ = Intensity
                    0EB5: get_char_damage_last_frame 19@ damager 2@ type 3@ part 4@ intensity 5@
                    0A96: 6@ = ped $PLAYER_ACTOR struct

                    if or
                        003B: 2@ == 6@
                        07D6: 2@ == $PLAYER_ACTOR
                    then
                        if
                            1@ <= 0
                        then
                            // Use 14@ for isolated pointer math and 6@ for priority lock
                            0085: 14@ = 31@
                            000A: 14@ += MEM_PRIORITY_LOCK
                            0A8D: 6@ = read_memory 14@ size 4 virtual_protect 0
                            if
                                6@ < 4
                            then
                                if
                                    3@ == 51
                                then
                                    0470: 3@ = actor $PLAYER_ACTOR current_weapon
                                    
                                    if
                                        3@ == 40
                                    then
                                        3@ = 39
                                    end

                                    if and
                                        8039: not 3@ == 16
                                        8039: not 3@ == 35
                                        8039: not 3@ == 36
                                        8039: not 3@ == 39
                                    then
                                        3@ = 35 
                                    end
                                end

                                11@ = 1
                                0085: 12@ = 3@
                            end
                        end
                    end
                end
            end
        end
    end
end
0ED0: return_script_event


// =========================================================================
// BAILOUT ROUTE
// =========================================================================
:ShionFF_Bailout
0A93: terminate_this_custom_script
