{$CLEO .c}
{$USE CLEO+}
{$USE memory}
{$USE ini}

0000: NOP
script_name "Sh_Gear"

wait 1000

// =========================================================================
// YORIGAMI MASTER HEAP DICTIONARY (PRUNED FOR SHION GEAR)
// =========================================================================
const
    MEM_SAVE_SLOT = 0
    MEM_PRIORITY_LOCK = 20
    MEM_SHION_HANDLE = 116
    MEM_SHION_STANCE = 120
    MEM_SHION_WEAPON_ID = 124
    MEM_SHION_AMMO_COUNT = 128
    MEM_COMBAT_RES_MODE = 132
    MEM_JOON_STASH_WEAPON = 136
    MEM_JOON_STASH_AMMO = 140
end

:Connect_To_Brain
wait 0
0AAA: 31@ = get_script_struct_named 'Y_Brain'
if
    31@ == 0 
then
    goto @Connect_To_Brain
end
000A: 31@ += 0x3C 
0A8D: 31@ = read_memory 31@ size 4 virtual_protect 0 
// 31@ NOW HOLDS THE MASTER POINTER.

// Initialize persistent state registers
0@ = -999  // Current Save Slot (from 0E2C)
1@ = -999  // Last Tracked Save Slot
2@ = 89    // MainKey (Y)
3@ = 82    // SwapKey (R)
4@ = 1     // CombatResourceMode (0 = Wallet Drain, 1 = Gear Swap)
5@ = 0     // Shion Stored Weapon ID (RAM)
6@ = 0     // Shion Stored Ammo Count (RAM)
7@ = 0     // Out of Ammo Audio/Subtitle Latch Flag
8@ = 0     // Empty-Ammo Debounce Accumulator Timer (ms)
9@ = 0     // Loading Screen Watchdog Flag

// Read Settings from Shion.ini
gosub @Reload_Gear_Settings

// Load active save slot data on initial boot
0E2C: 0@ = get_current_save_slot
if
    0@ < 0
then
    0@ = 0
end
gosub @Load_Weapon_Slot_From_Ini

:ShionGear_Loop
wait 50

// Detect loading screens or player despawning to catch save reloads
if
    8256: not player $PLAYER_CHAR defined
then
    9@ = 1
    goto @ShionGear_Loop
end

// Force slot reload from disk after any loading screen
if
    9@ == 1
then
    9@ = 0
    1@ = -999
end

// Check if Save Slot Changed or Save Loaded
0E2C: 0@ = get_current_save_slot
if
    0@ < 0
then
    0@ = 0
end

if
    803B: not 0@ == 1@
then
    gosub @Load_Weapon_Slot_From_Ini
end

// Read Live Handle from Shared Memory
0085: 30@ = 31@
000A: 30@ += MEM_SHION_HANDLE
0A8D: 10@ = read_memory 30@ size 4 virtual_protect 0

// If dismissed or in Hitodama form: sleep without touching actor opcodes
if
    10@ <= 0
then
    7@ = 0
    8@ = 0
    wait 250
    goto @ShionGear_Loop
end

// If physical actor is deleted or dead: sleep cleanly
if
    856D: not actor 10@ defined
then
    7@ = 0
    8@ = 0
    wait 250
    goto @ShionGear_Loop
end

if
    0118: actor 10@ dead
then
    7@ = 0
    8@ = 0
    wait 250
    goto @ShionGear_Loop
end

// Read Live Resource Mode from Shion.ini
0AF0: 4@ = read_int_from_ini_file "cleo\Shion.ini" section "Companion" key "CombatResourceMode"
0085: 30@ = 31@
000A: 30@ += MEM_COMBAT_RES_MODE
0A8C: write_memory 30@ size 4 value 4@ virtual_protect 0 // Broadcast mode (1 = Gear Swap, 0 = Wallet Drain)

// If Wallet Drain is active (Mode 0), Gear Swap stays dormant
if
    4@ == 0
then
    wait 250
    goto @ShionGear_Loop
end

// =========================================================================
// REAL-TIME AMMO MONITOR & EMPTY WEAPON FALLBACK WATCHDOG
// =========================================================================
0085: 30@ = 31@
000A: 30@ += MEM_SHION_STANCE
0A8D: 18@ = read_memory 30@ size 4 virtual_protect 0 // Stance State (0 = Peaceful/Concealed, 1 = Hostile/Armed)

0085: 30@ = 31@
000A: 30@ += MEM_SHION_WEAPON_ID
0A8D: 5@ = read_memory 30@ size 4 virtual_protect 0  // Keep synced with RAM

0085: 30@ = 31@
000A: 30@ += MEM_SHION_AMMO_COUNT
0A8D: 6@ = read_memory 30@ size 4 virtual_protect 0  // Keep synced with RAM

if and
    18@ == 1 // Only monitor ammo while actively in Hostile stance
    10@ > 0  // Solid Ped only
    5@ > 1   // Shion has a firearm assigned
then
    041A: 11@ = actor 10@ weapon 5@ ammo

    // Protect RAM: only update ammo count if greater than 0
    if
        11@ > 0
    then
        0085: 6@ = 11@
        0085: 30@ = 31@
        000A: 30@ += MEM_SHION_AMMO_COUNT
        0A8C: write_memory 30@ size 4 value 6@ virtual_protect 0
        8@ = 0
    else
        // If ammo reports 0, ensure she is not actively firing or cycling a round
        if
            82E0: not actor 10@ firing_weapon
        then
            000A: 8@ += 50
            // Must continuously report 0 ammo for 1.5s while not firing to be truly empty
            if
                8@ >= 1500
            then
                8@ = 0
                if
                    7@ == 0
                then
                    7@ = 1
                    gosub @Execute_Shion_Out_Of_Ammo
                end
            end
        else
            8@ = 0
        end
    end
else
    7@ = 0
    8@ = 0
end

// =========================================================================
// GEAR SWAP INTERACTION CONTROLLER (Y + R on foot within 4m)
// =========================================================================
01F5: 28@ = get_player_actor $PLAYER_CHAR
if and
    10@ > 0
    80DF: not actor 28@ driving
    8449: not actor 10@ in_a_car
    816B: not fading
then
    if
        8E25: not is_on_cutscene
    then
        if and
            0AB0: is_key_pressed 2@ // Y Key (ModifierKey)
            0AB0: is_key_pressed 3@ // R Key (SwapKey)
        then
            00A0: store_actor 28@ position_to 20@ 21@ 22@
            00A0: store_actor 10@ position_to 23@ 24@ 25@
            050A: 26@ = distance_between_XYZ 20@ 21@ 22@ and_XYZ 23@ 24@ 25@

            if
                8021: not 26@ > 4.0
            then
                :Wait_Gear_Swap_Release
                wait 50
                if
                    8256: not player $PLAYER_CHAR defined
                then
                    goto @ShionGear_Loop
                end
                if or
                    0AB0: is_key_pressed 2@
                    0AB0: is_key_pressed 3@
                then
                    goto @Wait_Gear_Swap_Release
                end

                // Only allow swap if Sister Talk (Tier 4) is not holding the screen
                0085: 30@ = 31@
                000A: 30@ += MEM_PRIORITY_LOCK
                0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
                if
                    19@ < 4
                then
                    gosub @Execute_Weapon_Swap
                    :Wait_Post_Swap_Release
                    wait 50
                    if or
                        0AB0: is_key_pressed 2@
                        0AB0: is_key_pressed 3@
                    then
                        goto @Wait_Post_Swap_Release
                    end
                end
            end
        end
    end
end

goto @ShionGear_Loop


// =========================================================================
// SUBROUTINE: EXECUTE WEAPON SWAP
// =========================================================================
:Execute_Weapon_Swap
01F5: 28@ = get_player_actor $PLAYER_CHAR
0470: 27@ = actor 28@ current_weapon

// Sync Shion's stored weapon & ammo from RAM
0085: 30@ = 31@
000A: 30@ += MEM_SHION_WEAPON_ID
0A8D: 5@ = read_memory 30@ size 4 virtual_protect 0

0085: 30@ = 31@
000A: 30@ += MEM_SHION_AMMO_COUNT
0A8D: 6@ = read_memory 30@ size 4 virtual_protect 0

0085: 30@ = 31@
000A: 30@ += MEM_SHION_STANCE
0A8D: 18@ = read_memory 30@ size 4 virtual_protect 0

// If Shion is currently in Hostile mode, ensure RAM has her live ammo
if and
    18@ == 1
    5@ > 1
    10@ > 0
then
    041A: 11@ = actor 10@ weapon 5@ ammo
    if
        11@ > 0
    then
        0085: 6@ = 11@
        0085: 30@ = 31@
        000A: 30@ += MEM_SHION_AMMO_COUNT
        0A8C: write_memory 30@ size 4 value 6@ virtual_protect 0
    end
end

// Case: Jo'on is unarmed (fists or knuckles) -> Retrieve Shion's weapon & pool ammo
if
    27@ <= 1
then
    if or
        5@ <= 1
        6@ <= 0
    then
        018C: play_sound 1057 at 0.0 0.0 0.0
        03EB: clear_small_messages_only
        0ADF: add_text_label "Y_SWNO" text "~b~Shion: ~s~\"There's nothing in my hands, Jo'on... you never gave me a gun.\""
        09C1: add_next_text_to_brief_history 1
        00BC: show_text_highpriority GXT 'Y_SWNO' time 2500 flag 1
        return
    end

    // Check if Jo'on already carries ammo for this exact weapon on her person
    041A: 29@ = actor 28@ weapon 5@ ammo
    if
        29@ > 0
    then
        005A: 6@ += 29@ // Pool Jo'on's existing ammo with Shion's weapon
    else
        // Check same-slot sibling weapons if Jo'on holds a sibling variant
        gosub @Check_And_Pool_Player_Sibling_Ammo
    end

    // Strip physical weapon from Shion ped completely
    048F: actor 10@ remove_weapons
    0555: remove_weapon 5@ from_actor 10@
    01B9: set_actor 10@ armed_weapon_to 0

    // Remove any weapon remnants in that slot from Jo'on first to prevent stacking
    0555: remove_weapon 5@ from_actor 28@

    // Give weapon and pooled stored ammo to Jo'on
    0781: get_weapon_with_ID 5@ model_to 12@
    if
        12@ > 0
    then
        0247: load_model 12@
        038B: load_requested_models

        :Wait_Player_Swap_Model
        wait 0
        if
            8256: not player $PLAYER_CHAR defined
        then
            0249: release_model 12@
            return
        end
        if
            8248: not model 12@ available
        then
            goto @Wait_Player_Swap_Model
        end

        01B2: give_actor 28@ weapon 5@ ammo 6@
        01B9: set_actor 28@ armed_weapon_to 5@
        017B: set_actor 28@ weapon 5@ ammo_to 6@
        0249: release_model 12@
    end

    // Reset Shion's designated weapon and ammo in RAM
    5@ = 0
    6@ = 0
    0085: 30@ = 31@
    000A: 30@ += MEM_SHION_WEAPON_ID
    0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

    0085: 30@ = 31@
    000A: 30@ += MEM_SHION_AMMO_COUNT
    0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

    // Put Shion into Peaceful Stance
    0085: 30@ = 31@
    000A: 30@ += MEM_SHION_STANCE
    0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
    gosub @Apply_Shion_Peaceful_Mode

    // Play Retrieve Dialogue
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8C: write_memory 30@ size 4 value 3 virtual_protect 0
    018C: play_sound 1058 at 0.0 0.0 0.0
    03EB: clear_small_messages_only
    0ADF: add_text_label "Y_SWRE" text "~y~Jo'on: ~s~\"Hand it over! Lemme show ya how to shoot!\"~n~~b~Shion: ~w~\"Fine by me... it was rattling my wrists anyway.\""
    09C1: add_next_text_to_brief_history 1
    00BC: show_text_highpriority GXT 'Y_SWRE' time 3000 flag 1
    wait 2500
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
    if
        19@ == 3
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
    end
    return
end

// Case: Jo'on is armed with a firearm (Pistols through Heavy Weapons: IDs 22 to 38)
if
    27@ > 1
then
    // Reject melee weapons, thrown explosives, and special tools/gadgets
    if or
        27@ < 22
        27@ > 38
    then
        018C: play_sound 1057 at 0.0 0.0 0.0
        return
    end

    041A: 29@ = actor 28@ weapon 27@ ammo
    if
        29@ <= 0
    then
        018C: play_sound 1057 at 0.0 0.0 0.0
        03EB: clear_small_messages_only
        0ADF: add_text_label "Y_SWNE" text "~y~Jo'on: ~s~\"Tch, it's bone-dry! What am I supposed to do, chuck it at 'em?!\""
        09C1: add_next_text_to_brief_history 1
        00BC: show_text_highpriority GXT 'Y_SWNE' time 2500 flag 1
        return
    end

    // Sub-case: Shion already holds a different weapon
    if and
        5@ > 1
        6@ > 0
        803B: not 27@ == 5@
    then
        018C: play_sound 1057 at 0.0 0.0 0.0
        03EB: clear_small_messages_only
        0ADF: add_text_label "Y_SWDF" text "~b~Shion: ~s~\"I've only got two hands, Jo'on... take this one back before I drop both.\""
        09C1: add_next_text_to_brief_history 1
        00BC: show_text_highpriority GXT 'Y_SWDF' time 3000 flag 1
        return
    end

    // Sub-case: Shion has the same weapon -> pool ammo from Jo'on to Shion
    if and
        5@ > 1
        6@ > 0
        003B: 27@ == 5@
    then
        // Strip Jo'on's gun completely
        0555: remove_weapon 27@ from_actor 28@
        01B9: set_actor 28@ armed_weapon_to 0

        // Pool ammo into Shion's reserve in RAM
        005A: 6@ += 29@
        0085: 30@ = 31@
        000A: 30@ += MEM_SHION_AMMO_COUNT
        0A8C: write_memory 30@ size 4 value 6@ virtual_protect 0

        // If Shion is currently hostile, update her physical actor ammo to match
        if and
            18@ == 1
            10@ > 0
        then
            017B: set_actor 10@ weapon 5@ ammo_to 6@
        end

        // Play Ammo Pooling Feedback
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8C: write_memory 30@ size 4 value 3 virtual_protect 0
        018C: play_sound 1058 at 0.0 0.0 0.0
        03EB: clear_small_messages_only
        0ADF: add_text_label "Y_SWPL" text "~y~Jo'on: ~s~\"Catch! Keep firin' away!\"~n~~b~Shion: ~w~\"There goes our dinner in smoke...\""
        09C1: add_next_text_to_brief_history 1
        00BC: show_text_highpriority GXT 'Y_SWPL' time 3000 flag 1
        wait 2500
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
        if
            19@ == 3
        then
            0085: 30@ = 31@
            000A: 30@ += MEM_PRIORITY_LOCK
            0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
        end
        return
    end

    // Sub-case: Initial weapon handover to Shion
    // Strip weapon from Jo'on
    0555: remove_weapon 27@ from_actor 28@
    01B9: set_actor 28@ armed_weapon_to 0

    // Assign to Shion in RAM
    0085: 5@ = 27@
    0085: 6@ = 29@
    0085: 30@ = 31@
    000A: 30@ += MEM_SHION_WEAPON_ID
    0A8C: write_memory 30@ size 4 value 5@ virtual_protect 0

    0085: 30@ = 31@
    000A: 30@ += MEM_SHION_AMMO_COUNT
    0A8C: write_memory 30@ size 4 value 6@ virtual_protect 0

    // Clear physical weapons on Shion
    048F: actor 10@ remove_weapons
    01B9: set_actor 10@ armed_weapon_to 0

    // Give weapon to Shion
    0781: get_weapon_with_ID 5@ model_to 12@
    if
        12@ > 0
    then
        0247: load_model 12@
        038B: load_requested_models

        :Wait_Shion_Give_Model
        wait 0
        if
            8256: not player $PLAYER_CHAR defined
        then
            0249: release_model 12@
            return
        end
        if
            856D: not actor 10@ defined
        then
            0249: release_model 12@
            return
        end
        if
            0118: actor 10@ dead
        then
            0249: release_model 12@
            return
        end
        if
            8248: not model 12@ available
        then
            goto @Wait_Shion_Give_Model
        end

        01B2: give_actor 10@ weapon 5@ ammo 6@
        01B9: set_actor 10@ armed_weapon_to 5@
        017B: set_actor 10@ weapon 5@ ammo_to 6@
        0249: release_model 12@
    end

    // Engage Hostile Stance
    0085: 30@ = 31@
    000A: 30@ += MEM_SHION_STANCE
    0A8C: write_memory 30@ size 4 value 1 virtual_protect 0
    gosub @Apply_Shion_Hostile_Mode

    // Play Handover Dialogue
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8C: write_memory 30@ size 4 value 3 virtual_protect 0
    018C: play_sound 1058 at 0.0 0.0 0.0
    03EB: clear_small_messages_only
    0ADF: add_text_label "Y_SWGV" text "~y~Jo'on: ~s~\"Hold this and cover me! And don't you dare drop it!\"~n~~b~Shion: ~w~\"Yeah yeah... I'll try not to miss.\""
    09C1: add_next_text_to_brief_history 1
    00BC: show_text_highpriority GXT 'Y_SWGV' time 3000 flag 1
    wait 2500
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
    if
        19@ == 3
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
    end
    return
end
return

    5@ == 34
then

    if
        29@ > 0
    then
        005A: 6@ += 29@
        0555: remove_weapon 33 from_actor 28@
        return
    end
end

if
    5@ == 33
then
    041A: 29@ = actor 28@ weapon 34 ammo
    if
        29@ > 0
    then
        005A: 6@ += 29@
        0555: remove_weapon 34 from_actor 28@
        return
    end
end

// Assault Rifles: M4 (31) <-> AK-47 (30)
if
    5@ == 31
then
    041A: 29@ = actor 28@ weapon 30 ammo
    if
        29@ > 0
    then
        005A: 6@ += 29@
        0555: remove_weapon 30 from_actor 28@
        return
    end
end

if
    5@ == 30
then
    041A: 29@ = actor 28@ weapon 31 ammo
    if
        29@ > 0
    then
        005A: 6@ += 29@

    end
end

// SMGs: MP5 (29) <-> Micro-Uzi (28) <-> Tec-9 (32)
if or
    5@ == 28

then
    041A: 29@ = actor 28@ weapon 29 ammo

        29@ > 0

        005A: 6@ += 29@

    if

    then

// =========================================================================
// SUBROUTINE: OUT OF AMMO TRIGGER
// =========================================================================
:Execute_Shion_Out_Of_Ammo
// Strip weapon
048F: actor 10@ remove_weapons
01B9: set_actor 10@ armed_weapon_to 0

// Clear RAM trackers
5@ = 0
6@ = 0
0085: 30@ = 31@
000A: 30@ += MEM_SHION_WEAPON_ID
0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

0085: 30@ = 31@
000A: 30@ += MEM_SHION_AMMO_COUNT
0A8C: write_memory 30@ size 4 value 0 virtual_protect 0

// Revert stance to Peaceful
0085: 30@ = 31@
000A: 30@ += MEM_SHION_STANCE

// Play Out of Ammo vocal alert (Tier 3 Action)
0085: 30@ = 31@

0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
if
    19@ < 4
then
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8C: write_memory 30@ size 4 value 3 virtual_protect 0
    018C: play_sound 1057 at 0.0 0.0 0.0
    03EB: clear_small_messages_only
    0ADF: add_text_label "Y_SWMT" text "~b~Shion: ~s~\"-Click-click-... it's empty, Jo'on.\"~n~~y~Jo'on: ~w~\"Tch! Get behind me, then!\""
    09C1: add_next_text_to_brief_history 1
    00BC: show_text_highpriority GXT 'Y_SWMT' time 3200 flag 1
    wait 3000
    0085: 30@ = 31@
    000A: 30@ += MEM_PRIORITY_LOCK
    0A8D: 19@ = read_memory 30@ size 4 virtual_protect 0
    if
        19@ == 3
    then
        0085: 30@ = 31@
        000A: 30@ += MEM_PRIORITY_LOCK
        0A8C: write_memory 30@ size 4 value 0 virtual_protect 0
    end
end
return
